## Задание 03 — статический вызов

```bash
# мы в: ~/lab1/linux/SP_lib
g++ -fPIC -shared SP.cpp -o SP.so
#  -fPIC    — код грузится по любому адресу (обязательно для .so)
#  -shared  — делаем библиотеку
#  -o       — имя выходного файла

# мы в: ~/lab1/linux/SP_03
cp ../SP_lib/SP.so .
g++ main.cpp SP.so -o SP_03
#  SP.so   — линкуем библиотеку (файл, не -lSP: искал бы libSP.so)

# запускаем (LD_LIBRARY_PATH — папка, где лежит библиотека)
LD_LIBRARY_PATH=. ./SP_03
```

**Ожидаем:** `sum = 15`, `sub = 5`, `mul = 50`, `div = 2`

**Показать преподавателю разницу:**

```bash
readelf -d SP_03 | grep SP.so
#  NEEDED  SP.so  — библиотека прописана в бинарнике, ОС грузит сама
```

## Задание 04 — динамический вызов (dlopen)

```bash
# мы в: ~/lab1/linux/SP_lib
g++ -fPIC -shared SP.cpp -o SP.so

# мы в: ~/lab1/linux/SP_04
cp ../SP_lib/SP.so .
g++ main.cpp -ldl -o SP_04
#  -ldl   — библиотека с dlopen/dlsym (в libc её нет)
#  Обрати внимание: SP.so в команде НЕТ — проект о ней не знает

# запускаем (LD_LIBRARY_PATH не нужен)
./SP_04
```

**Ожидаем:** `sum = 15`, `sub = 5`, `mul = 50`, `div = 2`

**Показать преподавателю разницу:**

```bash
readelf -d SP_04 | grep SP.so
#  пусто! библиотеки SP.so в бинарнике нет — грузится в runtime
```

## Задание 07 — C#, статический вызов (DllImport)

```bash
export DOTNET_ROOT="$HOME/.dotnet"
export PATH="$HOME/.dotnet:$PATH"
#  настройки dotnet, если открыт новый терминал

# мы в: ~/lab1/linux/cs/SP_07
dotnet publish -c Release -r linux-arm64 --self-contained true -p:PublishSingleFile=true
#  -c Release    — без отладки
#  -r linux-arm64 — под ARM-процессор
#  --self-contained — .NET внутри файла, отдельно ставить не надо
#  -p:PublishSingleFile=true — всё в один файл

# мы в: ~/lab1/linux/cs/SP_07/bin/Release/net8.0/linux-arm64/publish
cp ~/lab1/linux/SP_lib/SP.so .
ls -lh SP_07 SP.so
#  библиотека должна лежать РЯДО с программой, иначе DllNotFoundException

./SP_07
```

**Ожидаем:** `Сложение = 15`, `Вычитание = 5`, `Умножение = 50`, `Деление = 2`

## Всё сразу — собрать и запустить все четыре

```bash
export DOTNET_ROOT="$HOME/.dotnet"; export PATH="$HOME/.dotnet:$PATH"
L=~/Documents/VisualStudioCodeRepos/SystemProgramming/lab1
cd $L

# --- библиотека одна на все задания ---
(cd linux/SP_lib && g++ -fPIC -shared SP.cpp -o SP.so) && echo "SP.so собрана"

# --- 03: статический вызов ---
(cd linux/SP_03 && cp ../SP_lib/SP.so . && g++ main.cpp SP.so -o SP_03 \
  && LD_LIBRARY_PATH=. ./SP_03)

# --- 04: динамический вызов ---
(cd linux/SP_04 && cp ../SP_lib/SP.so . && g++ main.cpp -ldl -o SP_04 && ./SP_04)

# --- 07: C#, DllImport ---
(cd linux/cs/SP_07 && dotnet publish -c Release -r linux-arm64 --self-contained true \
   -p:PublishSingleFile=true -v q --nologo \
 && cd bin/Release/net8.0/linux-arm64/publish && cp $L/linux/SP_lib/SP.so . && ./SP_07)

# --- 08: C#, dlopen ---
(cd linux/cs/SP_08 && dotnet publish -c Release -r linux-arm64 --self-contained true \
   -p:PublishSingleFile=true -v q --nologo \
 && cd bin/Release/net8.0/linux-arm64/publish && cp $L/linux/SP_lib/SP.so . && ./SP_08)
```

**Ожидаем:** у всех четырёх `15`, `5`, `50`, `2`.
